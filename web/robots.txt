# Ratchet — https://ratchetgate.com
User-agent: *
Allow: /
# The console and the API are per-workspace and behind auth; crawling them
# produces nothing useful and generates rate-limited noise.
Disallow: /console
Disallow: /v1/
# …except the integration guide, which is public, static, and the whole point:
# an agent or a crawler that finds it can wire this up without a person.
Allow: /v1/integrate
Allow: /v1/vendors
Disallow: /mcp
# The OAuth consent screen collects a credential. It is not content, nobody
# should reach it from search, and an indexed password form on a young domain
# reads to automated classifiers as phishing.
Disallow: /oauth/
